{{- $image := .Page.Resources.GetMatch (printf "%s" (.Destination | safeURL)) -}} {{- $Permalink := .Destination | relURL | safeURL -}} {{- $alt := .PlainText | safeHTML -}} {{- $Width := 0 -}} {{- $Height := 0 -}} {{- $Srcset := "" -}} {{/* SVG and external images won't work with gallery layout, because their width and height attributes are unknown */}} {{- $galleryImage := false -}} {{- $destLower := lower .Destination -}} {{- $isRaster := or (strings.HasSuffix $destLower ".jpg") (strings.HasSuffix $destLower ".jpeg") (strings.HasSuffix $destLower ".png") -}} {{- $webpDestination := replaceRE `(?i)\.(jpe?g|png)$` ".webp" .Destination -}} {{- if $image -}} {{- $notSVG := ne (path.Ext .Destination) ".svg" -}} {{- $Permalink = $image.RelPermalink -}} {{- if $notSVG -}} {{- $Width = $image.Width -}} {{- $Height = $image.Height -}} {{- $galleryImage = true -}} {{- if (default true .Page.Site.Params.imageProcessing.content.enabled) -}} {{- $small := $image.Resize `480x` -}} {{- $big := $image.Resize `1024x` -}} {{- $Srcset = printf `%s 480w, %s 1024w` $small.RelPermalink $big.RelPermalink -}} {{- end -}} {{- end -}} {{- if $isRaster -}} {{- $webpResource := .Page.Resources.GetMatch (printf "%s" ($webpDestination | safeURL)) -}} {{- if $webpResource -}} {{- $Permalink = $webpResource.RelPermalink -}} {{- if (default true .Page.Site.Params.imageProcessing.content.enabled) -}} {{- $webpSmall := $webpResource.Resize `480x` -}} {{- $webpBig := $webpResource.Resize `1024x` -}} {{- $Srcset = printf `%s 480w, %s 1024w` $webpSmall.RelPermalink $webpBig.RelPermalink -}} {{- else -}} {{- $Srcset = "" -}} {{- end -}} {{- end -}} {{- end -}} {{- else -}} {{- if $isRaster -}} {{- $webpStaticPath := path.Join "static" (strings.TrimPrefix "/" $webpDestination) -}} {{- if fileExists $webpStaticPath -}} {{- $Permalink = $webpDestination | relURL | safeURL -}} {{- $Srcset = "" -}} {{- end -}} {{- end -}} {{- end -}}
